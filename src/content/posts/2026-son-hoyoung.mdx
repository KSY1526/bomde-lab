export const years = [d.first, d.first + 1, d.last];
export const f3 = (v) => v.toFixed(3).replace(/^0(?=\.)/, '');
export const pitches = ['포심', '슬라이더', '커브', '체인지업', '포크', '투심', '커터', '스위퍼'];
export const counts = ['초구', '타자 유리', '그 외', '2스트라이크'];
export const cntLabel = (c) => (c === '타자 유리' ? '타자가 유리한 카운트' : c);
export const yc = (y) => (y === d.last ? 'cur' : '');
export const bp = (p, y) => d.by_pitch.find((r) => r.pitch === p && r.season === y);
export const sp = (arr, g, y) => arr.find((r) => r.grp === g && r.season === y);
export const sh = (h, y) => d.slider_hand.find((r) => r.hand === h && r.season === y);
export const cnt = (arr, c, y) => arr.find((r) => r.cnt === c && r.season === y);
// gap > 0 = 타자 우세(붉은색), gap < 0 = 투수 우세(파란색). 기준(우르크 100)에서 10씩 멀어질 때마다 한 단계 진해진다
export const heat = (gap, step) => (gap === 0 ? null : { cls: gap > 0 ? 'up' : 'down', pct: [12, 24, 38, 52][Math.min(3, Math.floor((Math.abs(gap) - 1e-9) / step))] });
// 2스트라이크 표 전용: 모두 100 아래라 파란색은 그대로 두고, 진하기는 같은 해 리그 우타자 평균(같은 구종 묶음)보다 얼마나 낮은지로 정한다
export const tdVsLg = (r, lg, y) => {
  const gap = r.wrc_plus - lg.wrc_plus;
  const pct = gap >= 0 ? 6 : [12, 24, 38, 52][Math.min(3, Math.floor((-gap - 1e-9) / 15))];
  const small = r.pa < d.small_pa;
  return <td class={[yc(y), small ? 'small' : 'down'].filter(Boolean).join(' ')} style={small ? undefined : `--heat:${pct}%`}>{Math.round(r.wrc_plus)} ({r.pa})</td>;
};
export const tdLgBase = (r, y) => <td class={[yc(y), 'down'].filter(Boolean).join(' ')} style="--heat:6%">{Math.round(r.wrc_plus)} ({r.pa})</td>;
export const wrcTd = (r, y) => {
  if (!r) return <td class={yc(y)}>-</td>;
  const small = r.pa < d.small_pa;
  const h = heat(Math.round(r.wrc_plus - 100), 10);
  const cls = [yc(y), small ? 'small' : h ? h.cls : ''].filter(Boolean).join(' ');
  return <td class={cls} style={!small && h ? `--heat:${h.pct}%` : undefined}>{Math.round(r.wrc_plus)} ({r.pa})</td>;
};

## 상대는 직구 승부를 줄였다

상대 투수도 손호영에게 포심을 덜 던졌다. 손호영이 받은 공 중 포심 비율은 2024년 40.7%에서 2025년 37.5%, 올해 36.3%로 줄었다. 리그 우타자 평균(41.7%, 41.9%, 38.9%)보다 매년 낮고 2025년에는 4.4%p 차이가 났다.

볼이 스트라이크보다 많은, 타자가 유리한 카운트에서 차이가 가장 크다. 투수가 카운트를 잡아야 해서 포심을 많이 던지는 카운트인데 손호영에게 던진 포심 비율은 2024년 53.1%에서 올해 37.9%로 15%p 넘게 줄었다. 리그 우타자 평균은 49.7%에서 46.5%로 3%p 줄었을 뿐이다.

## 2스트라이크 이후에는 직구도 못 쳤다

2스트라이크 이후 OPS는 2024년 .610에서 2025년 .405, 올해 .322로 떨어졌다. 2024년에는 리그 우타자 평균 .554보다 높았지만 2025년과 올해는 리그 평균(.523, .529)에 한참 못 미친다. 2스트라이크 이후 삼진율도 35.3%에서 43.8%로 올랐다.

2스트라이크 이후에는 포심에도 대응하지 못했다. 2스트라이크 이후 포심으로 끝난 타석의 우르크는 2024년 52로 리그 우타자 평균 60과 비슷했지만 2025년 25(리그 평균 58), 올해 -64(27타석, 리그 평균 63)로 떨어졌다. 2025년에는 2스트라이크 포심에 방망이를 대는 비율이 88.8%로 리그 우타자 평균 82.0%보다 높았는데도 장타율이 .172에 그쳤다. 올해는 2스트라이크 이후 존 밖 포심에 방망이를 낸 비율이 55.6%로 리그 우타자 평균 40.7%보다 15%p 높다.

## 변화구를 보완한다면

상대는 타자가 유리한 카운트에서 손호영에게 포심을 주지 않았고 포심 외 공으로 카운트를 잡은 뒤 2스트라이크 승부로 끌고 갔다. 필자는 직구에 강한 손호영을 상대 배터리가 제대로 공략했다고 생각한다.

그래도 직구 강점은 확실하다. 2024년 포심 상대 우르크 139는 리그 우타자 평균보다 20 높았고 전체 우르크가 55로 떨어진 올해도 포심 상대로는 114를 기록했다. 2027년에는 변화구, 특히 우투수 슬라이더 약점을 보완해 반등하길 기대해 본다.
